import ifj26.*

def factorial(n: Int): Int =
 var result = 1
 var i: Int = n // uvedení typu je nepovinné
 while i > 1 do
   result = result * i
   i = i - 1
 result // poslední výraz v těle funkce určí návratovou hodnotu
